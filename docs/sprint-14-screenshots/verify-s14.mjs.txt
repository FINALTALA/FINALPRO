async function api(path, opts = {}) {
  const res = await fetch(`${API}${path}`, {
    method: opts.method ?? "GET",
    headers: {
      "Content-Type": "application/json",
      ...(opts.token ? { Authorization: `Bearer ${opts.token}` } : {}),
    },
    body: opts.body ? JSON.stringify(opts.body) : undefined,
  });
  const text = await res.text();
  return { status: res.status, body: text ? JSON.parse(text) : {} };
}

function otpFor(phone) {
  const text = fs.readFileSync(LOG, "utf8");
  const re = new RegExp(`OTP for ${phone.replace("+", "\\+")}: (\\d{6})`, "g");
  let m;
  let last = null;
  while ((m = re.exec(text))) last = m[1];
  return last;
}
async function waitOtp(phone) {
  for (let i = 0; i < 40; i++) {
    const c = otpFor(phone);
    if (c) return c;
    await new Promise((r) => setTimeout(r, 250));
  }
  throw new Error("no OTP for " + phone);
}

async function go(p, url) {
  try {
    await p.goto(url, { waitUntil: "networkidle" });
  } catch {
    await p.waitForTimeout(800);
    await p.goto(url, { waitUntil: "networkidle" });
  }
}

async function clear(name, locator) {
  await locator.first().scrollIntoViewIfNeeded();
  await mp.evaluate(() => window.scrollTo(0, document.documentElement.scrollHeight));
  await mp.waitForTimeout(250);
  const m = await locator.first().evaluate((node) => {
    const r = node.getBoundingClientRect();
    const nav = document.querySelector(".bottom-nav").getBoundingClientRect();
    const hit = document.elementFromPoint(r.left + r.width / 2, r.top + r.height / 2);
    return { bottom: r.bottom, navTop: nav.top, ok: r.bottom <= nav.top && (hit === node || node.contains(hit)) };
  });
  let clickable = true;
  try {
    await locator.first().click({ trial: true, timeout: 3000 });
  } catch {
    clickable = false;
  }
  record(`I mobile: ${name} is visible, above the bar, clickable`, m.ok && clickable, JSON.stringify(m));
  const overflow = await mp.evaluate(() => document.documentElement.scrollWidth - document.documentElement.clientWidth);
  record(`I mobile: no horizontal overflow on the page of ${name}`, overflow <= 0, String(overflow));
}
