import { chromium } from "playwright";
import { createRequire } from "module";

const require = createRequire("/workspace/package.json");
const { Client } = require("pg");

const API = "http://localhost:3011/api/v1";
const WEB = "http://localhost:3010";

const db = new Client({ connectionString: process.env.DATABASE_URL_PLAIN });
await db.connect();
const results = [];
function record(name, ok, extra = "") {
  results.push({ name, ok });
  console.log(ok ? "PASS" : "FAIL", name, ok ? "" : extra);
}
async function api(path, opts = {}) {
  const res = await fetch(`${API}${path}`, {
    method: opts.method ?? "GET",
    headers: {
      "Content-Type": "application/json",
      ...(opts.token ? { Authorization: `Bearer ${opts.token}` } : {}),
      ...(opts.idem ? { "Idempotency-Key": opts.idem } : {}),
    },
    body: opts.body ? JSON.stringify(opts.body) : undefined,
  });
  const text = await res.text();
  return { status: res.status, body: text ? JSON.parse(text) : {} };
}

let browser;
try {
  // Known state: no leftover holds/cart lines, full stock.
  for (const t of ["checkout_reservation_slots", "checkout_reservation_items", "checkout_reservations"]) await db.query(`delete from ${t}`);
  await db.query(`delete from cart_items`);
  await db.query(`update branch_stock set quantity=25, "reservedQuantity"=0`);

  const login = (await api("/auth/login", { method: "POST", body: { phone: "+970560000003", password: "DemoCustomer#2026" } })).body.session_token;
  const card = (await api("/discovery/all")).body.items[0];
  const slug = card.cheapest_offer.vendor_slug;
  const offer = (await api(`/storefronts/${slug}/offers/${card.cheapest_offer.offer_id}`)).body;
  const variantId = offer.variants[0].id;
  const vendorId = offer.vendor_id;
  const branchA = (await db.query(`select id from store_branches where "vendorId"=$1 and name='الفرع الرئيسي'`, [vendorId])).rows[0].id;

    await page.waitForURL("**/checkout");

  }

  await addLine(1);

  record("1 physical branch defaults to pick-up", (await methodSelect().inputValue()) === "PICKUP");

  const options = await methodSelect().locator("option").allInnerTexts();

  await page.getByRole("button", { name: "إلغاء والعودة" }).first().click();

  await db.query(`delete from delivery_window_exceptions where "windowId" in (select id from delivery_windows where "branchId"=$1)`, [branchA]);
  await db.query(`delete from delivery_windows where "branchId"=$1`, [branchA]);

  const warning = await page.getByText("لا توجد طريقة استلام أو توصيل متاحة لهذا الفرع").count();

  await db.query(`insert into branch_stock (id,"vendorId","branchId","offerVariantId",quantity,"reservedQuantity","updatedAt") values (gen_random_uuid(),$1,$2,$3,5,0,now()) on conflict ("branchId","offerVariantId") do update set quantity=5`, [vendorId, branchB, variantId]);

  const proceedBlocked = await line.locator('input[type="checkbox"]').isDisabled();
